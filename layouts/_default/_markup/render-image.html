{{- /* 图片渲染钩子： 1) 把 Obsidian 粘贴生成的 static/xxx 或 /static/xxx 路径改成站点根路径 /xxx； 2) 统一加 loading=lazy + decoding=async，外链图片也走同一套。 这样在 Obsidian 里直接粘贴图片（存到 static/images/obsidian/）就能正常显示，不用手改路径。 */ -}} {{- $u := .Destination -}} {{- if hasPrefix $u "static/" -}}{{- $u = strings.TrimPrefix "static" $u -}}{{- end -}} {{- if hasPrefix $u "/static/" -}}{{- $u = strings.TrimPrefix "/static" $u -}}{{- end -}}
